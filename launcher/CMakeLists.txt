cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(BakuganLauncher LANGUAGES CXX)
endif()
if(NOT WIN32)
    message(FATAL_ERROR "The launcher currently supports Windows only")
endif()
file(READ "${CMAKE_CURRENT_LIST_DIR}/../compatibility.json" COMPATIBILITY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/../compatibility.json")
string(JSON XEX_HASH GET "${COMPATIBILITY}" xex sha256)
string(JSON XEX_SIZE GET "${COMPATIBILITY}" xex size)
string(JSON PROJECT_VERSION_TEXT GET "${COMPATIBILITY}" project_version)
string(JSON FILE_COUNT LENGTH "${COMPATIBILITY}" required_files)
math(EXPR FILE_LAST "${FILE_COUNT} - 1")
set(REQUIRED_FILES "")
foreach(INDEX RANGE 0 ${FILE_LAST})
    string(JSON NAME GET "${COMPATIBILITY}" required_files ${INDEX} name)
    string(JSON SIZE GET "${COMPATIBILITY}" required_files ${INDEX} size)
    string(APPEND REQUIRED_FILES "  {L\"${NAME}\", ${SIZE}ULL},\n")
endforeach()
configure_file(compatibility.h.in compatibility.h @ONLY)
enable_language(RC)
add_executable(BakuganLauncher WIN32 main.cpp ../resources/app.rc)
target_include_directories(BakuganLauncher PRIVATE ../resources)
target_compile_features(BakuganLauncher PRIVATE cxx_std_20)
target_compile_definitions(BakuganLauncher PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(BakuganLauncher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(BakuganLauncher PRIVATE bcrypt ole32 shell32 user32 gdi32 uuid)
set_target_properties(BakuganLauncher PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(MSVC)
    target_compile_options(BakuganLauncher PRIVATE /W4 /utf-8)
endif()
